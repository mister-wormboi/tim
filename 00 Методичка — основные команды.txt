1.1. Установка ОС и обновление
Скачайте ISO Ubuntu с ubuntu.com/download/desktop, проверьте SHA-256 по сайту.
В ВМ создайте диск от 60 ГБ; для Android и Unity желательно от 100 ГБ и 32 ГБ

В VirtualBox/VMware установите гостевые дополнения; для KVM — guest agent.
Для Android внутри ВМ требуется вложенная виртуализация; иначе используйте
физический телефон. Установка ОС на реальный диск стирает выбранные разделы.

--- НАЧАЛО PYTHON ---
from http.server import BaseHTTPRequestHandler, HTTPServer
import json

class Handler(BaseHTTPRequestHandler):
    def reply(self, status, data):
        body = json.dumps(data).encode('utf-8')
        self.send_response(status)
        self.send_header('Content-Type', 'application/json; charset=utf-8')
        self.send_header('Content-Length', str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        if self.path == '/health':
            self.reply(200, {'status': 'ok'})
        else:
            self.reply(404, {'error': 'not found'})

    def do_POST(self):
        if self.path != '/echo':
            self.reply(404, {'error': 'not found'})
            return
        try:
            size = int(self.headers.get('Content-Length', '0'))
            if not 0 < size <= 65536:
                self.reply(400, {'error': 'invalid size'})
                return
            data = json.loads(self.rfile.read(size))
        except (ValueError, UnicodeDecodeError):
            self.reply(400, {'error': 'invalid JSON'})
            return
        self.reply(200, data)

HTTPServer(('127.0.0.1', 8000), Handler).serve_forever()

const { app, BrowserWindow } = require('electron');
function createWindow() {
  const win = new BrowserWindow({
    width: 800, height: 600,
    webPreferences: { nodeIntegration: false, contextIsolation: true }
  });
  win.loadFile('index.html');
}
app.whenReady().then(() => {
  createWindow();
  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) createWindow();
  });
});
app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') app.quit();
});

8. КРАТКИЙ ПОРЯДОК ДЕМОНСТРАЦИИ И ОЦЕНКИ

Общее для всех шести: ОС/ядро/драйвер; выбранный сетевой интерфейс и HTTPS;

приложение по заданию; тест результата; совместимость; руководство пользователя.
Фиксируйте только реально выполненные проверки. Не выдавайте пример команды
или наличие установленного пакета за доказательство настроенной функции.

Ориентир на 80 минут при заранее подготовленных ISO, SDK и кэше пакетов:
0–15 мин — ОС, обновления, драйверы; 15–25 — сеть, SSH, общий доступ;
25–40 — программы варианта и базовое ПО; 40–55 — защита, права и аудит;
55–65 — резервные копии и снимки; 65–75 — контрольный пример и совместимость;
75–80 — проверка документов и демонстрация. Загрузка больших SDK и создание
полного образа могут занять больше времени; подготовьте ресурсы заранее.
